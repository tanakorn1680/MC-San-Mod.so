cmake_minimum_required(VERSION 3.22)
project(mcbridge_mc LANGUAGES CXX)

# Same build recipe as LiteLDev/LeviLaunchroid's examples/full-cpp-mod (Apache-2.0): the preloader SDK (headers + the preloader
# target we link against) comes from a checkout of https://github.com/LiteLDev/preloader-android.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(PRELOADER_ANDROID_ROOT "${CMAKE_CURRENT_LIST_DIR}/../preloader-android" CACHE PATH "Path to LiteLDev/preloader-android")
if(NOT EXISTS "${PRELOADER_ANDROID_ROOT}/include/pl/Mod.hpp")
    message(FATAL_ERROR "PRELOADER_ANDROID_ROOT must point to a preloader-android checkout (missing include/pl/Mod.hpp in ${PRELOADER_ANDROID_ROOT})")
endif()
if(NOT ANDROID)
    message(FATAL_ERROR "This mod only builds with the Android NDK toolchain")
endif()

include(FetchContent)
set(MC_DEPS)
if(NOT TARGET nlohmann_json::nlohmann_json)
    FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
    list(APPEND MC_DEPS nlohmann_json)
endif()
if(NOT TARGET Boost::pfr)
    FetchContent_Declare(boost_pfr GIT_REPOSITORY https://github.com/boostorg/pfr.git GIT_TAG boost-1.88.0)
    list(APPEND MC_DEPS boost_pfr)
endif()
if(NOT TARGET magic_enum::magic_enum)
    FetchContent_Declare(magic_enum GIT_REPOSITORY https://github.com/Neargye/magic_enum.git GIT_TAG v0.9.7)
    list(APPEND MC_DEPS magic_enum)
endif()
if(NOT TARGET fmt::fmt)
    FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.2.0)
    set(FMT_DOC OFF CACHE BOOL "" FORCE)
    set(FMT_TEST OFF CACHE BOOL "" FORCE)
    set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
    list(APPEND MC_DEPS fmt)
endif()
FetchContent_MakeAvailable(${MC_DEPS})

if(NOT TARGET preloader)
    add_subdirectory("${PRELOADER_ANDROID_ROOT}" "${CMAKE_BINARY_DIR}/_preloader_android")
endif()

set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out/${ANDROID_ABI}")
find_library(MC_LOG_LIB log)

add_library(mcbridge_mc SHARED src/McBridgeMod.cpp)
set_target_properties(mcbridge_mc PROPERTIES OUTPUT_NAME "mcbridge_mc")
target_include_directories(mcbridge_mc PRIVATE "${CMAKE_CURRENT_LIST_DIR}/src" "${PRELOADER_ANDROID_ROOT}/include")
target_link_libraries(mcbridge_mc PRIVATE nlohmann_json::nlohmann_json Boost::pfr magic_enum::magic_enum fmt::fmt preloader)
if(MC_LOG_LIB)
    target_link_libraries(mcbridge_mc PRIVATE "${MC_LOG_LIB}")
endif()
target_compile_options(mcbridge_mc PRIVATE
    $<$<COMPILE_LANG_AND_ID:CXX,Clang,GNU>:-Wall -Wextra -fvisibility=hidden -fvisibility-inlines-hidden>)
